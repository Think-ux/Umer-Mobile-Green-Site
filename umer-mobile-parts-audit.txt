export type OrderType = "Customer" | "Retailer";

export type OrderStatus =
  | "Pending"
  | "Processing"
  | "Packed"
  | "Dispatched"
  | "Delivered"
  | "Cancelled";

export type SavedOrder = {
  orderNumber: string;
  orderType: OrderType;
  customer: {
    name: string;
    phone: string;
    address: string;
    city: string;
    payment: string;
  };
  items: {
    product: {
      id: number;
      name: string;
      price: number;
      slug?: string;
      image?: string;
    };
    quantity: number;
    unitPrice: number;
  }[];
  subtotal: number;
  delivery: number;
  total: number;
  status: OrderStatus;
  createdAt: string;
};

export const ORDERS_KEY = "umer-orders";
export const LAST_ORDER_KEY = "umer-last-order";
export const ORDERS_EVENT = "umer-orders-changed";
export const LAST_ORDER_EVENT = "umer-last-order-changed";

export function createOrderNumber() {
  return "UMP-" + Date.now().toString().slice(-8);
}

export function nowIso() {
  return new Date().toISOString();
}

export function readOrders(): SavedOrder[] {
  if (typeof window === "undefined") return [];

  try {
    const value = JSON.parse(
      localStorage.getItem(ORDERS_KEY) || "[]"
    );

    return Array.isArray(value) ? value as SavedOrder[] : [];
  } catch {
    return [];
  }
}

export function saveOrders(orders: SavedOrder[]) {
  localStorage.setItem(ORDERS_KEY, JSON.stringify(orders));
  window.dispatchEvent(new Event(ORDERS_EVENT));
}

export function updateSavedOrder(
  orderNumber: string,
  status: OrderStatus
) {
  const orders = readOrders().map((order) =>
    order.orderNumber === orderNumber
      ? { ...order, status }
      : order
  );

  saveOrders(orders);

  const lastRaw = localStorage.getItem(LAST_ORDER_KEY);

  if (lastRaw) {
    try {
      const last = JSON.parse(lastRaw);

      if (last.orderNumber === orderNumber) {
        localStorage.setItem(
          LAST_ORDER_KEY,
          JSON.stringify({ ...last, status })
        );
      }
    } catch {
      // Ignore malformed demo data.
    }
  }

  return orders;
}

"use client";

import {
  createContext,
  useCallback,
  useContext,
  useMemo,
  useSyncExternalStore,
  type ReactNode,
} from "react";

import type { Product } from "@/data/store";
import {
  emptySnapshot,
  readStorage,
  subscribeToStorage,
  writeStorage,
} from "@/lib/storage";

export type CartProduct = Product;

export type CartItem = {
  product: Product;
  quantity: number;
};

export type CartContextType = {
  items: CartItem[];
  count: number;
  itemCount: number;
  subtotal: number;
  delivery: number;
  total: number;
  addToCart: (product: Product, quantity?: number) => void;
  removeFromCart: (productId: number) => void;
  updateQuantity: (productId: number, quantity: number) => void;
  clearCart: () => void;
};

const STORAGE_KEY = "umer-mobile-parts-cart";
const CART_EVENT = "umer-cart-changed";

const CartContext = createContext<CartContextType | undefined>(undefined);

function isCartItem(item: unknown): item is CartItem {
  if (!item || typeof item !== "object") return false;

  const value = item as {
    product?: unknown;
    quantity?: unknown;
  };

  return (
    !!value.product &&
    typeof value.product === "object" &&
    typeof value.quantity === "number" &&
    value.quantity > 0
  );
}

function parseCart(raw: string): CartItem[] {
  if (!raw) {
    return [];
  }

  try {
    const parsed: unknown = JSON.parse(raw);
    return Array.isArray(parsed) ? parsed.filter(isCartItem) : [];
  } catch {
    return [];
  }
}

function getCartSnapshot() {
  return readStorage(STORAGE_KEY);
}

function persistCart(items: CartItem[]) {
  writeStorage(STORAGE_KEY, JSON.stringify(items), CART_EVENT);
}

const subscribeCart = subscribeToStorage(CART_EVENT);

export function CartProvider({ children }: { children: ReactNode }) {
  const raw = useSyncExternalStore(
    subscribeCart,
    getCartSnapshot,
    emptySnapshot
  );

  const items = useMemo(() => parseCart(raw), [raw]);

  const addToCart = useCallback((product: Product, quantity = 1) => {
    const safeQuantity = Math.max(1, Math.min(quantity, product.stock));
    const current = parseCart(getCartSnapshot());
    const existing = current.find((item) => item.product.id === product.id);

    if (!existing) {
      persistCart([...current, { product, quantity: safeQuantity }]);
      return;
    }

    persistCart(
      current.map((item) =>
        item.product.id === product.id
          ? {
              ...item,
              quantity: Math.min(product.stock, item.quantity + safeQuantity),
            }
          : item
      )
    );
  }, []);

  const removeFromCart = useCallback((productId: number) => {
    persistCart(
      parseCart(getCartSnapshot()).filter(
        (item) => item.product.id !== productId
      )
    );
  }, []);

  const updateQuantity = useCallback(
    (productId: number, quantity: number) => {
      if (quantity <= 0) {
        removeFromCart(productId);
        return;
      }

      persistCart(
        parseCart(getCartSnapshot()).map((item) =>
          item.product.id === productId
            ? {
                ...item,
                quantity: Math.min(item.product.stock, quantity),
              }
            : item
        )
      );
    },
    [removeFromCart]
  );

  const clearCart = useCallback(() => {
    persistCart([]);
  }, []);

  const count = useMemo(
    () => items.reduce((sum, item) => sum + item.quantity, 0),
    [items]
  );

  const subtotal = useMemo(
    () =>
      items.reduce((sum, item) => sum + item.product.price * item.quantity, 0),
    [items]
  );

  const delivery = subtotal > 0 ? 250 : 0;
  const total = subtotal + delivery;

  const value = useMemo(
    () => ({
      items,
      count,
      itemCount: count,
      subtotal,
      delivery,
      total,
      addToCart,
      removeFromCart,
      updateQuantity,
      clearCart,
    }),
    [
      items,
      count,
      subtotal,
      delivery,
      total,
      addToCart,
      removeFromCart,
      updateQuantity,
      clearCart,
    ]
  );

  return (
    <CartContext.Provider value={value}>{children}</CartContext.Provider>
  );
}

export function useCart(): CartContextType {
  const context = useContext(CartContext);

  if (!context) {
    throw new Error("useCart must be used inside CartProvider");
  }

  return context;
}

"use client";

import Link from "next/link";
import { FormEvent, Suspense, useState } from "react";
import { useRouter, useSearchParams } from "next/navigation";
import { useCart } from "@/components/CartProvider";
import { products } from "@/data/store";
import {
  LAST_ORDER_EVENT,
  LAST_ORDER_KEY,
  ORDERS_EVENT,
  ORDERS_KEY,
  createOrderNumber,
  nowIso,
} from "@/lib/orders";
import { readStorage, writeStorage } from "@/lib/storage";

type CustomerForm = {
  name: string;
  phone: string;
  address: string;
  city: string;
  payment: string;
};

export default function CheckoutPage() {
  return (
    <Suspense fallback={<main className="p-8">Loading checkout...</main>}>
      <CheckoutForm />
    </Suspense>
  );
}

function CheckoutForm() {
  const { items: cartItems, clearCart } = useCart();
  const router = useRouter();
  const searchParams = useSearchParams();
  const orderType =
    searchParams.get("type") === "retailer" ? "retailer" : "customer";
  const productId = Number(searchParams.get("product"));
  const retailerProductId =
    Number.isInteger(productId) && productId > 0 ? productId : null;

  const selectedRetailerProduct =
    orderType === "retailer" && retailerProductId !== null
      ? products.find((product) => product.id === retailerProductId)
      : undefined;

  const items =
    orderType === "retailer" && selectedRetailerProduct
      ? [{
          product: {
            ...selectedRetailerProduct,
            price: Math.round(selectedRetailerProduct.price * 0.82),
          },
          quantity: 1,
        }]
      : orderType === "retailer"
        ? cartItems.map((item) => ({
            ...item,
            product: {
              ...item.product,
              price: Math.round(item.product.price * 0.82),
            },
          }))
        : cartItems;

  const subtotal = items.reduce(
    (sum, item) => sum + item.product.price * item.quantity,
    0
  );
  const delivery = subtotal > 0 ? 250 : 0;
  const total = subtotal + delivery;

  const [form, setForm] = useState<CustomerForm>({
    name: "",
    phone: "",
    address: "",
    city: "Lahore",
    payment: "Cash on Delivery",
  });

  const [error, setError] = useState("");
  const [placing, setPlacing] = useState(false);

  const [paymentReference, setPaymentReference] = useState("");
  function updateField(
    field: keyof CustomerForm,
    value: string
  ) {
    setForm((current) => ({
      ...current,
      [field]: value,
    }));

    setError("");
  }

  function submitOrder(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();

    if (placing) {
      return;
    }

    if (items.length === 0) {
      setError("Your cart is empty.");
      return;
    }

    const name = form.name.trim();
    const phone = form.phone.trim();
    const address = form.address.trim();

    if (name.length < 3) {
      setError("Please enter your full name.");
      return;
    }

    const phoneDigits = phone.replace(/\D/g, "");

    if (
      phoneDigits.length !== 11 ||
      !phoneDigits.startsWith("03")
    ) {
      setError("Please enter a valid Pakistani phone number, e.g. 03001234567.");
      return;
    }

    if (address.length < 10) {
      setError("Please enter your complete delivery address.");
      return;
    }

    setPlacing(true);

    const orderNumber = createOrderNumber();

    const order = {
      orderType,
      orderNumber,
      customer: {
        ...form,
        name,
        phone,
        address,
      },
      items,
      subtotal,
      delivery,
      total,
      status: "Pending",
      createdAt: nowIso(),
    };

    try {
      const existingOrdersRaw = readStorage(ORDERS_KEY);

      let existingOrders: unknown[] = [];

      if (existingOrdersRaw) {
        try {
          const parsed = JSON.parse(existingOrdersRaw);

          if (Array.isArray(parsed)) {
            existingOrders = parsed;
          }
        } catch {
          existingOrders = [];
        }
      }

      writeStorage(
        ORDERS_KEY,
        JSON.stringify([order, ...existingOrders]),
        ORDERS_EVENT
      );

      writeStorage(LAST_ORDER_KEY, JSON.stringify(order), LAST_ORDER_EVENT);

      if (orderType === "customer") clearCart();

      router.push(
        `/order-confirmation?order=${encodeURIComponent(orderNumber)}`
      );
    } catch {
      setPlacing(false);
      setError(
        "Unable to place the order right now. Please try again."
      );
    }
  }

  if (items.length === 0) {
    return (
      <main className="min-h-screen bg-slate-50 px-4 py-20">
        <div className="mx-auto max-w-xl rounded-3xl bg-white p-10 text-center shadow-sm">

          <div className="mx-auto flex h-16 w-16 items-center justify-center rounded-full bg-orange-50 text-2xl">
            Cart
          </div>

          <h1 className="mt-6 text-3xl font-black">
            Your cart is empty
          </h1>

          <p className="mt-3 text-slate-500">
            Add products before continuing to checkout.
          </p>

          <Link
            href="/"
            className="mt-7 inline-flex rounded-xl bg-orange-500 px-6 py-3 font-black text-white hover:bg-orange-600"
          >
            Start Shopping
          </Link>

        </div>
      </main>
    );
  }

  return (
    <main className="min-h-screen bg-slate-50 px-4 py-10">
      <div className="mx-auto max-w-7xl">

        <div className="mb-8">
          <p className="text-sm font-black uppercase tracking-widest text-orange-500">
            Checkout
          </p>

          <h1 className="mt-2 text-4xl font-black">
            Complete Your Order
          </h1>

          <p className="mt-2 text-slate-500">
            Enter your delivery details and choose a payment method.
          </p>
        </div>

        <form
          noValidate
          onSubmit={submitOrder}
          className="grid gap-8 lg:grid-cols-[1fr_360px]"
        >

          <div className="space-y-6">

            <section className="rounded-2xl border bg-white p-6 shadow-sm">
              <h2 className="text-xl font-black">
                Customer Information
              </h2>

              <div className="mt-5 grid gap-4 sm:grid-cols-2">

                <div>
                  <label className="text-sm font-bold">
                    Full Name
                  </label>

                  <input
                    value={form.name}
                    onChange={(event) =>
                      updateField("name", event.target.value)
                    }
                    placeholder="Your full name"
                    className="mt-2 w-full rounded-xl border px-4 py-3 outline-none focus:border-orange-500"
                  />
                </div>

                <div>
                  <label className="text-sm font-bold">
                    Phone Number
                  </label>

                  <input
                    inputMode="numeric"
                    maxLength={11}
                    value={form.phone}
                    onChange={(event) =>
                      updateField(
                        "phone",
                        event.target.value.replace(/\D/g, "").slice(0, 11)
                      )
                    }
                    placeholder="03001234567"
                    className="mt-2 w-full rounded-xl border px-4 py-3 outline-none focus:border-orange-500"
                  />
                </div>

                <div className="sm:col-span-2">
                  <label className="text-sm font-bold">
                    Delivery Address
                  </label>

                  <textarea
                    value={form.address}
                    onChange={(event) =>
                      updateField("address", event.target.value)
                    }
                    placeholder="House / Shop / Street / Area"
                    rows={4}
                    className="mt-2 w-full rounded-xl border px-4 py-3 outline-none focus:border-orange-500"
                  />
                </div>

                <div>
                  <label className="text-sm font-bold">
                    City
                  </label>

                  <select
                    value={form.city}
                    onChange={(event) =>
                      updateField("city", event.target.value)
                    }
                    className="mt-2 w-full rounded-xl border px-4 py-3 outline-none focus:border-orange-500"
                  >
                    <option>Lahore</option>
                    <option>Karachi</option>
                    <option>Islamabad</option>
                    <option>Rawalpindi</option>
                    <option>Faisalabad</option>
                    <option>Multan</option>
                    <option>Other</option>
                  </select>
                </div>

              </div>
            </section>

            <section className="rounded-2xl border bg-white p-6 shadow-sm">
  <h2 className="text-xl font-black">Payment Method</h2>

  <p className="mt-2 text-sm text-slate-500">
    Select your preferred payment method. Online payments require

  <div className="mt-5 space-y-3">
    {[

      "Bank Transfer",
      "JazzCash",
      "Easypaisa",
    ].map((method) => (
      <label
        key={method}

            ? "border-orange-500 bg-orange-50"
            : "border-slate-200 hover:border-slate-400"
        }`}
      >
        <input
          type="radio"
          name="payment"
          value={method}
          checked={form.payment === method}
          onChange={(event) => {
            updateField("payment", event.target.value);
            setPaymentReference("");
          }}
          className="accent-orange-500"
        />

        <span className="font-bold">{method}</span>

        <span className="ml-auto text-xs text-slate-500">
          {method === "Cash on Delivery" ? "Pay on delivery" : "Manual payment"}
        </span>
      </label>
    ))}
  </div>

  {form.payment === "Cash on Delivery" ? (
    <div className="mt-5 rounded-xl border border-green-200 bg-green-50 p-4">
      <p className="font-black text-green-800">

  ) : (
    <div className="mt-5 rounded-xl border bg-slate-50 p-5">
      <h3 className="font-black">{form.payment} Instructions</h3>

      <p className="mt-2 text-sm leading-6 text-slate-600">
